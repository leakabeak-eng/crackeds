		local function getWeaponClass(weaponName)
			local ok, result = pcall(GetWeaponProperties, weaponName)

		local function stringMatchesAny(str, list)
			for _, keyword in ipairs(list) do
				if str:find(keyword, 1, true) then
					return true
				end
			end
			return false
		end

		local KNIFE_KEYWORDS = { "Knife", "Bayonet", "Karambit" }
		local GLOVE_KEYWORDS = { "Glove", "Hand Wraps" }

		local function isKnife(weaponName)
			if getWeaponClass(weaponName) == "Melee" then
				return true
			end
			return type(weaponName) == "string" and (stringMatchesAny(weaponName, KNIFE_KEYWORDS) or weaponName == "LightSaber")
		end

		local function isGlove(weaponName)
			if getWeaponClass(weaponName) == "Glove" then
				return true
			end
			return type(weaponName) == "string" and stringMatchesAny(weaponName, GLOVE_KEYWORDS)
		end

		local function weaponAssetExists(weaponName)
			local weaponsFolder = ReplicatedStorage.Assets:FindFirstChild("Weapons")
			return weaponsFolder ~= nil and weaponsFolder:FindFirstChild(weaponName) ~= nil
		end

		local OWNER_PROPERTIES = { "Owner", "Player", "User", "PlayerOwner", "CharacterOwner" }
		local CHARACTER_PROPERTIES = { "Character", "Model", "CharacterModel", "Subject" }

		local function getOwnerPlayer(instance)
			if not instance then
				return nil
			end

			for _, propName in ipairs(OWNER_PROPERTIES) do
				local val = instance[propName]
				if typeof(val) == "Instance" and val:IsA("Player") then
					return val
				end
			end

			for _, propName in ipairs(CHARACTER_PROPERTIES) do
				local val = instance[propName]
				if typeof(val) == "Instance" and val:IsA("Model") then
					local player = Players:GetPlayerFromCharacter(val)
					if player then
						return player
					end
				end
			end

			local parent = instance.Parent
			local depth = 0
			while parent and depth < 5 do
				if typeof(parent) == "Instance" then
					if parent:IsA("Player") then
						return parent
					end
					if parent:IsA("Model") then
						local player = Players:GetPlayerFromCharacter(parent)
						if player then
							return player
						end
					end
				end
				parent = parent.Parent
				depth = depth + 1
			end

			return nil
		end

		local function getForcedWeaponAndSkin(weaponName, originalSkin)
			if type(weaponName) ~= "string" then
				return weaponName, originalSkin
			end
			local team = getLocalTeam()
			if not team then
				return weaponName, originalSkin
			end
			local forcedWeapon

			if isKnife(weaponName) then
				forcedWeapon = skinState.forcedKnife[team]
				if not (forcedWeapon and forcedWeapon ~= weaponName and weaponAssetExists(forcedWeapon)) then
					forcedWeapon = weaponName
				end
			elseif isGlove(weaponName) then
				forcedWeapon = skinState.forcedGlove[team]
				if not (forcedWeapon and forcedWeapon ~= weaponName and weaponAssetExists(forcedWeapon)) then
					forcedWeapon = weaponName
				end
			else
				forcedWeapon = weaponName
			end

			return forcedWeapon, skinState.forcedSkins[team][forcedWeapon] or originalSkin
		end

				local extraArgs = table.pack(...)

				local forcedWeapon, forcedSkin = getForcedWeaponAndSkin(weaponArg, skinArg)
				local okCall, callResult = pcall(originalSkinsHandlers[methodName], forcedWeapon, forcedSkin, ...)
				if okCall and callResult ~= nil then
					return callResult
				end
				return originalSkinsHandlers[methodName](weaponArg, skinArg, table.unpack(extraArgs, 1, extraArgs.n))
			end
		end

				return originalViewmodelNew(owner, weaponName, skinName, extraArg)
			end

				if isKnife(weaponName) then
					local forcedKnife = skinState.forcedKnife[currentTeam]
					if forcedKnife and forcedKnife ~= weaponName and weaponAssetExists(forcedKnife) then
						return originalViewmodelNew(owner, forcedKnife, skinState.forcedSkins[currentTeam][forcedKnife] or skinName, extraArg)
					end
				elseif isGlove(weaponName) then
					local forcedGlove = skinState.forcedGlove[currentTeam]

						return originalViewmodelNew(owner, forcedGlove, skinState.forcedSkins[currentTeam][forcedGlove] or skinName, extraArg)
					end
				end

				return originalViewmodelNew(owner, weaponName, skinName, extraArg)
			end)

			skinState.inLocalViewmodel = false
			if not okViewmodel then
				warn("[BSSkins] Viewmodel.new error:", viewmodelInstance)
				return originalViewmodelNew(owner, weaponName, skinName, extraArg)
			end
			return viewmodelInstance
		end

		local loadoutPlacements = {}

		local function registerPlacement(placement)
			local key
			if placement.kind == "options" then
				key = ("%s|options|%s|%d"):format(placement.team, tostring(placement.type), placement.index)
			else
				key = ("%s|equipped|%s"):format(placement.team, tostring(placement.path))
			end

			local existing = loadoutPlacementsByKey[key]
			if existing then
				existing.new = placement.new
				return
			end
			placement.key = key
			loadoutPlacementsByKey[key] = placement
			table.insert(loadoutPlacements, placement)
		end

		local function applyLoadoutPlacements(useNew)
			local loadoutData = DataController.Get(localPlayer, "Loadout")
			if not loadoutData then
				return
			end

			for _, placement in ipairs(loadoutPlacements) do
				local teamLoadout = loadoutData[placement.team]
				if teamLoadout then
					local targetId = useNew and placement.new or placement.prev
					if placement.kind == "options" then
						local options = teamLoadout.Loadout and teamLoadout.Loadout[placement.type] and teamLoadout.Loadout[placement.type].Options
						if options then
							options[placement.index] = targetId
						end
					elseif placement.kind == "equipped" then
						if teamLoadout.Equipped then
							teamLoadout.Equipped[placement.path] = targetId
						end
					end
				end
			end
		end

		local function handleEquipAction(actionName, payload)
			if not payload or not payload.Identifier then
				return false
			end
			local item = findSyntheticItem(payload.Identifier)
			if not item then
				return false
			end
			sanitizeItemData(item)
			local targetTeam = normalizeTeam(payload.Team) or getLocalTeam()
			if not targetTeam then
				return false
			end

			if item.Type == "Melee" then
				skinState.forcedKnife[targetTeam] = item.Name
			elseif item.Type == "Glove" then
				skinState.forcedGlove[targetTeam] = item.Name
			end

			skinState.forcedSkins[targetTeam][item.Name] = item.Skin
			local loadoutData = DataController.Get(localPlayer, "Loadout")

			if loadoutData and loadoutData[targetTeam] then
				local teamLoadout = loadoutData[targetTeam]

				if actionName == "EquipLoadoutSkin" and payload.Type then
					local options = teamLoadout.Loadout and teamLoadout.Loadout[payload.Type] and teamLoadout.Loadout[payload.Type].Options
					if options then
						local slotIndex = math.clamp((payload.Slot or 0) + 1, 1, math.max(#options, 1))
						registerPlacement({
							team = targetTeam,
							kind = "options",
							type = payload.Type,
							index = slotIndex,
							prev = options[slotIndex],
							new = item._id,
						})
						options[slotIndex] = item._id
					end
				elseif actionName == "EquipSpecialItem" and payload.Path then
					if teamLoadout.Equipped then
						registerPlacement({
							team = targetTeam,
							kind = "equipped",
							path = payload.Path,
							prev = teamLoadout.Equipped[payload.Path],
							new = item._id,
						})
						teamLoadout.Equipped[payload.Path] = item._id
					end
				end
			end

			pcall(function()
				InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
			end)

			debugLog(("Force [%s] %s | %s"):format(targetTeam, item.Name, item.Skin))
			return true
		end

		local hookedRemotesList = {}

		local function hookEquipRemote(actionName)
			local remoteNode = SecurityRemotes
			for _, key in ipairs({ "Inventory", actionName }) do
				if type(remoteNode) ~= "table" then
					return
				end
				remoteNode = remoteNode[key]
			end

			if type(remoteNode) ~= "table" or type(remoteNode.Send) ~= "function" then
				return
			end
			local originalSend = remoteNode.Send
			table.insert(hookedRemotesList, { node = remoteNode, original = originalSend })

			remoteNode.Send = function(payload)
				if skinState.enabled then
					local handled = false
					pcall(function()
						handled = handleEquipAction(actionName, payload)
					end)
					if handled then
						return
					end
				end
				return originalSend(payload)
			end
		end

		local function waitForDataLoaded()
			local elapsed = 0
			while not DataController.IsDataLoaded(localPlayer) do
				task.wait(0.1)
				elapsed = elapsed + 0.1
				if elapsed >= 30 then
					return false
				end
			end
			return true
		end

		local function fetchAvailableSkins()
			local elapsed = 0
			while elapsed < 15 do
				local attr = ReplicatedStorage:GetAttribute("AvaiableSkins")
				if attr then
					local ok, decoded = pcall(HttpService.JSONDecode, HttpService, attr)
					if ok and decoded and next(decoded) then
						return decoded
					end
				end
				task.wait(0.2)
				elapsed = elapsed + 0.2
			end
			return nil
		end

		local syntheticSkins = {}
		local cachedAvailableSkins = nil

		local function injectAvailableSkins(skinsData)
			local inventory = DataController.Get(localPlayer, "Inventory")
			if not inventory then
				return
			end
			local existingLookup = {}

			for _, item in ipairs(inventory) do
				if item.Name and item.Skin then
					existingLookup[item.Name .. "::" .. item.Skin] = true
				end
			end

			for _, item in pairs(syntheticSkins) do
				existingLookup[item.Name .. "::" .. item.Skin] = true
			end

			local injectedCount = 0

			for weaponName, skinGroup in pairs(skinsData) do
				for skinName, skinProperties in pairs(skinGroup) do
					if skinName ~= "Stock" and skinProperties.isEnabled ~= false then
						local identifier = weaponName .. "::" .. skinName
						if not existingLookup[identifier] then
							local syntheticItem = {
								_id = HttpService:GenerateGUID(false),
								Name = weaponName,
								Skin = skinName,
								Type = skinProperties.type or "Weapon",
								Float = 0,
								Pattern = 1,
								StatTrack = false,
								NameTag = false,
								Charm = false,
								Stickers = {},
								Rarity = skinProperties.rarity or "Mil-Spec",
								IsTradeable = false,
								Serial = nil,
								MetaData = {
									CreatedAt = 0,
									LastTradeAt = 0,
									TradeHistory = {},
									Owner = tostring(localPlayer.UserId),
									Origin = "unlockall",
								},
							}
							table.insert(inventory, syntheticItem)
							syntheticSkins[syntheticItem._id] = syntheticItem
							existingLookup[identifier] = true
							injectedCount = injectedCount + 1
						end
					end
				end
			end

			debugLog(("injected %d skins"):format(injectedCount))
		end

		local function removeSyntheticSkins()
			local inventory = DataController.Get(localPlayer, "Inventory")
			if not inventory then
				return
			end
			local removedCount = 0

			for i = #inventory, 1, -1 do
				local item = inventory[i]
				if item and item._id and syntheticSkins[item._id] then
					table.remove(inventory, i)
					removedCount = removedCount + 1
				end
			end

			if removedCount > 0 then
				pcall(function()
					InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
				end)
			end
		end

		local function restoreSyntheticSkins()
			local inventory = DataController.Get(localPlayer, "Inventory")
			if not inventory then
				return
			end
			local currentIds = {}

			for _, item in ipairs(inventory) do
				if item._id then
					currentIds[item._id] = true
				end
			end

			local addedCount = 0

			for id, item in pairs(syntheticSkins) do
				if not currentIds[id] then
					table.insert(inventory, item)
					addedCount = addedCount + 1
				end
			end

			if addedCount > 0 then
				pcall(function()
					InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
				end)
			end
		end

		local function toggleSkinChanger(enabled)
			local isEnabled = enabled and true or false
			if isEnabled == skinState.enabled then
				return
			end
			skinState.enabled = isEnabled

			if skinState.enabled then
				if next(syntheticSkins) then
					restoreSyntheticSkins()
				elseif cachedAvailableSkins then
					injectAvailableSkins(cachedAvailableSkins)
				end
				applyLoadoutPlacements(true)
				pcall(function()
					InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
				end)
				quickSwitchWeapon()
			else
				applyLoadoutPlacements(false)
				removeSyntheticSkins()
				pcall(function()
					InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
				end)
				quickSwitchWeapon()
			end

			debugLog(skinState.enabled and "enabled" or "disabled")
		end

		local isSkinDataReady = false

		task.spawn(function()
			if not waitForDataLoaded() then
				warn("[BSSkins] player data never loaded - UnlockAll inactive")
				return
			end
			cachedAvailableSkins = fetchAvailableSkins()
			if cachedAvailableSkins then
				injectAvailableSkins(cachedAvailableSkins)
				isSkinDataReady = true
				debugLog("data ready - toggle via GUI")
			else
				warn("[BSSkins] skin data unavailable - UnlockAll inactive")
			end
		end)

		local function applySavedPlacements(placements)
			if type(placements) ~= "table" then
				return false
			end
			if not isSkinDataReady then
				return false
			end
			local loadoutData = DataController.Get(localPlayer, "Loadout")
			if not loadoutData then
				return false
			end

			for _, placement in ipairs(placements) do
				local teamName = normalizeTeam(placement.team)
				if teamName and loadoutData[teamName] and placement.weapon and placement.skin then
					local matchedSynthetic = nil
					for _, item in ipairs(syntheticSkins) do
						if item.Name == placement.weapon and item.Skin == placement.skin then
							matchedSynthetic = item
							break
						end
					end

					if matchedSynthetic then
						local teamLoadout = loadoutData[teamName]
						if placement.kind == "options" and placement.type and placement.index then
							local options = teamLoadout.Loadout and teamLoadout.Loadout[placement.type] and teamLoadout.Loadout[placement.type].Options
							if options and options[placement.index] and options[placement.index] ~= matchedSynthetic._id then
								registerPlacement({
									team = teamName,
									kind = "options",
									type = placement.type,
									index = placement.index,
									prev = options[placement.index],
									new = matchedSynthetic._id,
								})
								options[placement.index] = matchedSynthetic._id
							end
						elseif placement.kind == "equipped" and placement.path then
							if teamLoadout.Equipped and teamLoadout.Equipped[placement.path] ~= matchedSynthetic._id then
								registerPlacement({
									team = teamName,
									kind = "equipped",
									path = placement.path,
									prev = teamLoadout.Equipped[placement.path],
									new = matchedSynthetic._id,
								})
								teamLoadout.Equipped[placement.path] = matchedSynthetic._id
							end
						end
					end
				end
			end

			pcall(function()
				InventoryController.OnInventoryChanged:Fire(DataController.Get(localPlayer, "Inventory"))
			end)

			return true
		end

		bsApi = {
			Force = function(weapon, skin, team)
				local targetTeam = normalizeTeam(team) or getLocalTeam()
				if targetTeam then
					skinState.forcedSkins[targetTeam][weapon] = skin
				end
				quickSwitchWeapon()
			end,
			ForceKnife = function(knife, team)
				local targetTeam = normalizeTeam(team) or getLocalTeam()
				if targetTeam then
					skinState.forcedKnife[targetTeam] = knife
				end
				quickSwitchWeapon()
			end,
			ForceGloves = function(gloves, team)
				local targetTeam = normalizeTeam(team) or getLocalTeam()
				if targetTeam then
					skinState.forcedGlove[targetTeam] = gloves
				end
				quickSwitchWeapon()
			end,
			Clear = function(weapon, team)
				local targetTeam = normalizeTeam(team) or getLocalTeam()
				if targetTeam then
					skinState.forcedSkins[targetTeam][weapon] = nil
				end
				quickSwitchWeapon()
			end,
			ClearAll = function()
				for _, team in ipairs(ALL_TEAMS) do
					skinState.forcedSkins[team] = {}
					skinState.forcedKnife[team] = nil
					skinState.forcedGlove[team] = nil
				end
				quickSwitchWeapon()
			end,
			Get = function(weapon, team)
				local targetTeam = normalizeTeam(team) or getLocalTeam()
				return targetTeam and skinState.forcedSkins[targetTeam][weapon] or nil
			end,
			GetForced = function()
				local result = { T = {}, CT = {}, placements = {} }
				for _, team in ipairs(ALL_TEAMS) do

						skinCopy[weapon] = skin
					end

						knife = skinState.forcedKnife[team],
						glove = skinState.forcedGlove[team],
					}
				end

							team = placement.team,

							skin = matchedItem.Skin,

					return

					end

				end)

		task.spawn(function()

				task.wait(0.2)
			end

			if isSkinDataReady and pendingPlacements then
				task.wait(0.3)
				local toApply = pendingPlacements
				pendingPlacements = nil
				pcall(applySavedPlacements, toApply)
			end
		end)
	end

local cachedGetPlayersOnTeam = nil

local function getPlayersOnTeamFunction()
	if cachedGetPlayersOnTeam then
		return cachedGetPlayersOnTeam
	end

	pcall(function()
		if typeof(filtergc) == "function" then
			cachedGetPlayersOnTeam = filtergc("function", { Name = "getPlayersOnTeam" }, true)
		end
	end)

	if not cachedGetPlayersOnTeam and typeof(getgc) == "function" then
		pcall(function()
			for _, fn in ipairs(getgc()) do
				if type(fn) == "function" then
					local info = debug.getinfo(fn)
					if info and info.name == "getPlayersOnTeam" then
						cachedGetPlayersOnTeam = fn
						break
					end
				end
			end
		end)
	end

	return cachedGetPlayersOnTeam
end

GetPlayersOnTeamFunc = getPlayersOnTeamFunction

local function getPlayerObjectName(entry)
	if type(entry) == "string" then
		return entry
	elseif type(entry) == "userdata" or typeof(entry) == "Instance" then
		return entry.Name
	end
	return nil
end

local function getLocalTeamList()
	local getTeamFunc = getPlayersOnTeamFunction()
	if not getTeamFunc then
		return nil
	end

	local okCT, ctPlayers = pcall(getTeamFunc, "Counter-Terrorists")
	local okT, tPlayers = pcall(getTeamFunc, "Terrorists")

	if okCT and type(ctPlayers) == "table" then
		for _, member in ipairs(ctPlayers) do
			if getPlayerObjectName(member) == localPlayer.Name then
				return ctPlayers
			end
		end
	end

	if okT and type(tPlayers) == "table" then
		for _, member in ipairs(tPlayers) do
			if getPlayerObjectName(member) == localPlayer.Name then
				return tPlayers
			end
		end
	end

	return nil
end

GetLocalTeamList = getLocalTeamList

local function isEnemyPlayer(player)
	if player == localPlayer then
		return false
	end
	local localTeamMembers = getLocalTeamList()
	if not localTeamMembers then
		return true
	end

	for _, member in ipairs(localTeamMembers) do
		if getPlayerObjectName(member) == player.Name then
			return false
		end
	end

	return true
end

IsEnemyPlayer = isEnemyPlayer

local function isAlive(charFolder)
	if not charFolder or not charFolder.Parent then
		return false
	end

	if charFolder:GetAttribute("Dead") == true then
		return false
	end

	if charFolder:GetAttribute("Invincible") == true then
		return false
	end

	local humanoid = charFolder:FindFirstChildOfClass("Humanoid")
	if humanoid and humanoid.Health <= 0 and humanoid.MaxHealth > 0 then
		return false
	end

	return true
end

IsAlive = isAlive

do
	local raycastParams = RaycastParams.new()
	raycastParams.FilterType = Enum.RaycastFilterType.Exclude
	raycastParams.IgnoreWater = true

	local visibilityCache = {}
	local CACHE_LIFETIME = 0.1

	isVisible = function(targetPart)
		local now = os.clock()
		local cached = visibilityCache[targetPart]
		if cached and now - cached.time < CACHE_LIFETIME then
			return cached.visible
		end

		local myChar = getCharacterFolder(localPlayer)
		if not myChar then
			return true
		end

		local originPart = myChar:FindFirstChild("HumanoidRootPart") or myChar:FindFirstChild("Head")
		if not originPart then
			return true
		end

		local direction = targetPart.Position - originPart.Position
		raycastParams.FilterDescendantsInstances = { myChar }

		local hitResult = Workspace:Raycast(originPart.Position, direction, raycastParams)
		local visible = not hitResult or hitResult.Instance:IsDescendantOf(targetPart.Parent)

		visibilityCache[targetPart] = { visible = visible, time = now }
		return visible
	end
end

local function swapSurfaces(targetModel, sourceModel)
	if not targetModel or not sourceModel then
		return
	end

	local surfaceAppearances = {}
	for _, child in ipairs(sourceModel:GetChildren()) do
		if child:IsA("SurfaceAppearance") then
			surfaceAppearances[child.Name] = child
		end
	end

	if not next(surfaceAppearances) then
		return
	end

	for _, descendant in ipairs(targetModel:GetDescendants()) do
		if descendant:IsA("MeshPart") and surfaceAppearances[descendant.Name] then
			for _, child in ipairs(descendant:GetChildren()) do
				if child:IsA("SurfaceAppearance") then
					child:Destroy()
				end
			end

			surfaceAppearances[descendant.Name]:Clone().Parent = descendant
		end
	end
end

local function untagName(name)
	return name:gsub(BACKUP_TAG .. "$", "")
end

local function tagName(name)
	if name:find(BACKUP_TAG .. "$") then
		return name
	end
	return name .. BACKUP_TAG
end

local weaponSlotBackups = {}

local function backupSlot(slotName)
	if weaponSlotBackups[slotName] then
		return
	end

	local backupData = {}
	local weaponModel = weaponsFolder:FindFirstChild(slotName)
	if weaponModel then
		local cameraModel = weaponModel:FindFirstChild("Camera")
		if cameraModel then
			backupData.Camera = cameraModel:Clone()
		end

		local characterModel = weaponModel:FindFirstChild("Character")
		if characterModel then
			backupData.Character = characterModel:Clone()
		end
	end

	local animFolder = weaponAnimationsFolder:FindFirstChild(slotName)
	if animFolder then
		backupData.Anims = animFolder:Clone()
	end

	weaponSlotBackups[slotName] = backupData
end

local function restoreRS(slotName)
	local backupData = weaponSlotBackups[slotName]
	if not backupData then
		return
	end

	local weaponModel = weaponsFolder:FindFirstChild(slotName)
	if weaponModel then
		if backupData.Camera then
			local oldCamera = weaponModel:FindFirstChild("Camera")
			if oldCamera then
				oldCamera:Destroy()
			end
			local cameraClone = backupData.Camera:Clone()
			cameraClone.Name = "Camera"
			cameraClone.Parent = weaponModel
		end

		if backupData.Character then
			local oldChar = weaponModel:FindFirstChild("Character")
			if oldChar then
				oldChar:Destroy()
			end
			local charClone = backupData.Character:Clone()
			charClone.Name = "Character"
			charClone.Parent = weaponModel
		end
	end

	if backupData.Anims then
		local oldAnims = weaponAnimationsFolder:FindFirstChild(slotName)
		if oldAnims then
			oldAnims:Destroy()
		end
		local animClone = backupData.Anims:Clone()
		animClone.Name = slotName
		animClone.Parent = weaponAnimationsFolder
	end
end

		for _, child in ipairs(customWeaponsFolder:GetChildren()) do

	assignKnifeToSlot = function(slotName, targetKnifeName)
		pcall(function()
			if not customWeaponsFolder then
				return
			end

			local previousKnife = assignedKnives[slotName]
			if previousKnife then
				local existing = customWeaponsFolder:FindFirstChild(slotName)
				if existing then
					existing.Name = tagName(previousKnife)
				end
				assignedKnives[slotName] = nil
			end

			if targetKnifeName == "Default" then
				local defaultModule = defaultKnifeModules[slotName]
				if defaultModule then
					local current = customWeaponsFolder:FindFirstChild(slotName)
					if current then
						current:Destroy()
					end
					local clone = defaultModule:Clone()
					clone.Name = slotName
					clone.Parent = customWeaponsFolder
				end
				return
			end

			local otherSlot = slotName == "CT Knife" and "T Knife" or "CT Knife"
			local otherKnife = assignedKnives[otherSlot]
			if otherKnife and otherKnife:lower() == targetKnifeName:lower() then
				assignKnifeToSlot(otherSlot, "Default")
			end

			local targetFolder = findKnifeFolder(targetKnifeName)
			if not targetFolder then
				return
			end

			local current = customWeaponsFolder:FindFirstChild(slotName)
			if current then
				current:Destroy()
			end

			targetFolder.Name = slotName
			assignedKnives[slotName] = targetKnifeName

			local sprayPatternsIgnore = { SprayPatterns = true }
			for _, child in ipairs(customWeaponsFolder:GetChildren()) do
				local name = child.Name
				if name ~= "CT Knife" and name ~= "T Knife" and not name:find(BACKUP_TAG .. "$") then
					if not isGun(name) and not ignoredWeaponTypes[name] and not sprayPatternsIgnore[untagName(name)] then
						child.Name = tagName(name)
					end
				end
			end
		end)
	end

local function patchAnimator(targetModel, animMap)
	if not targetModel or not next(animMap) then
		return
	end

	local animController = targetModel:FindFirstChildOfClass("AnimationController")
	local animator = animController and animController:FindFirstChildOfClass("Animator")

	if not animator then
		for _, descendant in ipairs(targetModel:GetDescendants()) do
			if descendant:IsA("Animator") then
				animator = descendant
				break
			end
		end
	end

	if not animator then
		return
	end

	for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
		pcall(function()
			track:Stop(0)
		end)
	end

	local loadedTracks = {}
	for name, id in pairs(animMap) do
		local animation = Instance.new("Animation")
		animation.AnimationId = id
		animation.Name = name

		local ok, track = pcall(function()
			return animator:LoadAnimation(animation)
		end)

		if ok and track then
			loadedTracks[name] = track
		end
		animation:Destroy()
	end

	if loadedTracks.Idle then
		pcall(function()
			loadedTracks.Idle:Play()
		end)
	end
end

local function buildGunSkinList(gunName)
	local gunFolder = skinsFolder:FindFirstChild(gunName)
	local displayNames = { "Default" }
	local fullKeys = { "Default" }
	if not gunFolder then
		return displayNames, fullKeys
	end

	local skinNames = {}
	for _, child in ipairs(gunFolder:GetChildren()) do
		if child:IsA("Folder") and child.Name ~= "Default" and child.Name ~= "None" then
			table.insert(skinNames, child.Name)
		end
	end

	table.sort(skinNames)
	for _, skinName in ipairs(skinNames) do
		table.insert(displayNames, skinName)
		table.insert(fullKeys, gunName .. "|" .. skinName)
	end

	return displayNames, fullKeys
end

local function buildKnifeSkinList()
	local displayNames = { "Default" }
	local fullKeys = { "Default" }
	local knifeToSkinsMap = {}

	for _, child in ipairs(skinsFolder:GetChildren()) do
		local name = child.Name
		if not (isGun(name) or ignoredWeaponTypes[name]) then
			if not (name:find("Glove") or name:find("Wraps")) then
				for _, subFolder in ipairs(child:GetChildren()) do
					if subFolder:IsA("Folder") and subFolder.Name ~= "Default" and subFolder.Name ~= "None" then
						if not knifeToSkinsMap[subFolder.Name] then
							knifeToSkinsMap[subFolder.Name] = {}
						end

						local exists = false
						for _, knife in ipairs(knifeToSkinsMap[subFolder.Name]) do
							if knife == name then
								exists = true
								break
							end
						end

						if not exists then
							table.insert(knifeToSkinsMap[subFolder.Name], name)
						end
					end
				end
			end
		end
	end

	local sortedSkinNames = {}
	for skinName in pairs(knifeToSkinsMap) do
		table.insert(sortedSkinNames, skinName)
	end
	table.sort(sortedSkinNames)

	for _, skinName in ipairs(sortedSkinNames) do
		local knifeList = knifeToSkinsMap[skinName]
		table.sort(knifeList)
		for _, knifeName in ipairs(knifeList) do
			table.insert(displayNames, skinName .. " (" .. knifeName .. ")")
			table.insert(fullKeys, knifeName .. "|" .. skinName)
		end
	end

	return displayNames, fullKeys
end

local charmsConfig = {
	ArmEnabled = false,
	ArmColor1 = Color3.fromRGB(19, 0, 255),
	ArmColor2 = Color3.fromRGB(0, 0, 0),
	ArmSpeed = 2,
	ArmTransparency = 0,
	ArmMaterial = "ForceField",
	ArmDisable = false,
	GunEnabled = false,
	GunColor1 = Color3.fromRGB(0, 0, 0),
	GunColor2 = Color3.fromRGB(19, 0, 255),

do
	local charmState = {
		armPhase = 0,
		gunPhase = 0,
		connection = nil,
		originalProperties = {},
	}

	local function charmLerpColor(c1, c2, alpha)
		return Color3.new(
			c1.R + (c2.R - c1.R) * alpha,
			c1.G + (c2.G - c1.G) * alpha,
			c1.B + (c2.B - c1.B) * alpha
		)
	end

	local function charmGradColor(phase, c1, c2)
		return charmLerpColor(c1, c2, (math.sin(phase) + 1) * 0.5)
	end

	local IGNORED_ARM_KEYWORDS = {
		"hitbox",
		"collision",
		"trigger",
		"proxy",
		"handguard",
		"handle",
		"foregrip",
		"charm",
		"armor",
	}

	local ARM_KEYWORDS = {
		"arm",
		"hand",
		"finger",
		"wrist",
		"palm",
		"thumb",
		"glove",
		"sleeve",
		"leftarm",
		"rightarm",

		"rightupperarm",
		"leftlowerarm",
		"rightlowerarm",

		"rightforearm",
	}

		"muzzle",
		"flash",
		"effect",
		"particle",
		"emitter",
		"attachment",
	}

	charmIsArm = function(part)
		if not part or not part:IsA("BasePart") then
			return false
		end
		local lowerName = part.Name:lower()

		for _, keyword in ipairs(IGNORED_ARM_KEYWORDS) do
			if lowerName:find(keyword, 1, true) then
				return false
			end
		end

		for _, keyword in ipairs(ARM_KEYWORDS) do
			if lowerName:find(keyword, 1, true) then
				return true
			end
		end

		return false
	end

	charmIsValid = function(part)
		if not part or not part:IsA("BasePart") then
			return false
		end

		if part.Transparency >= 0.99 then
			return false
		end
		local lowerName = part.Name:lower()

		for _, keyword in ipairs(IGNORED_GUN_KEYWORDS) do
			if lowerName:find(keyword, 1, true) then
				return false
			end
		end

		return true
	end

	getVMParts = function()
		local armParts = {}
		local gunParts = {}
		local viewModel = currentCamera:FindFirstChildOfClass("Model")
		if not viewModel then
			return armParts, gunParts
		end

		for _, descendant in ipairs(viewModel:GetDescendants()) do
			if descendant:IsA("BasePart") and charmIsValid(descendant) then
				if charmIsArm(descendant) then
					table.insert(armParts, descendant)
				else
					table.insert(gunParts, descendant)
				end
			end
		end

		return armParts, gunParts
	end

	charmSaveOrig = function(part)
		if not charmState.originalProperties[part] then
			charmState.originalProperties[part] = {
				Material = part.Material,
				Color = part.Color,
				Transparency = part.Transparency,
				Reflectance = part.Reflectance,
				TextureID = part:IsA("MeshPart") and part.TextureID or nil,
			}
		end
	end

	charmRestoreAll = function()
		for part, origProps in pairs(charmState.originalProperties) do
			if part and part.Parent then
				pcall(function()
					part.Material = origProps.Material
					part.Color = origProps.Color
					part.Transparency = origProps.Transparency
					part.Reflectance = origProps.Reflectance or 0
					if origProps.TextureID ~= nil then
						part.TextureID = origProps.TextureID
					end
				end)
			end
		end
		charmState.originalProperties = {}
	end

	charmApply = function(partsList, color, material, transparency)
		for _, part in ipairs(partsList) do
			if charmIsValid(part) then
				charmSaveOrig(part)
				pcall(function()
					part.Material = material
					part.Color = color
					part.Transparency = transparency
					part.Reflectance = 0
					part.LocalTransparencyModifier = 0

					if part:IsA("MeshPart") then
						part.TextureID = ""
					end

					for _, child in ipairs(part:GetChildren()) do
						if child:IsA("Texture") or child:IsA("Decal") or child:IsA("SurfaceAppearance") then
							child:Destroy()
						end
					end
				end)
			end
		end
	end

	charmRestoreParts = function(partsList)
		for _, part in ipairs(partsList) do
			local origProps = charmState.originalProperties[part]
			if origProps and part and part.Parent then
				pcall(function()
					part.Material = origProps.Material
					part.Color = origProps.Color
					part.Transparency = origProps.Transparency
					part.Reflectance = origProps.Reflectance or 0
					if origProps.TextureID ~= nil then
						part.TextureID = origProps.TextureID
					end
				end)
				charmState.originalProperties[part] = nil
			end
		end
	end

	charmUpdate = function(deltaTime)
		if not charmsConfig.ArmEnabled and not charmsConfig.GunEnabled then
			if charmState.connection then
				charmState.connection:Disconnect()
				charmState.connection = nil
			end
			charmRestoreAll()
			return
		end

		charmState.armPhase = charmState.armPhase + deltaTime * charmsConfig.ArmSpeed
		charmState.gunPhase = charmState.gunPhase + deltaTime * charmsConfig.GunSpeed

		local armColor = charmGradColor(charmState.armPhase, charmsConfig.ArmColor1, charmsConfig.ArmColor2)
		local gunColor = charmGradColor(charmState.gunPhase, charmsConfig.GunColor1, charmsConfig.GunColor2)
		local armMaterial = Enum.Material[charmsConfig.ArmMaterial] or Enum.Material.ForceField
		local gunMaterial = Enum.Material[charmsConfig.GunMaterial] or Enum.Material.ForceField
		local armParts, gunParts = getVMParts()

		if charmsConfig.ArmEnabled then
			charmApply(armParts, armColor, armMaterial, charmsConfig.ArmTransparency)
		else
			charmRestoreParts(armParts)
		end

		if charmsConfig.GunEnabled then
			charmApply(gunParts, gunColor, gunMaterial, charmsConfig.GunTransparency)
		else
			charmRestoreParts(gunParts)
		end
	end

	charmStart = function()
		if charmState.connection then
			charmState.connection:Disconnect()
			charmState.connection = nil
		end

		if charmsConfig.ArmEnabled or charmsConfig.GunEnabled then
			charmState.connection = addConnection(RunService.RenderStepped:Connect(charmUpdate))
		else
			charmRestoreAll()
		end
	end

	charmForceRefresh = function()
		if charmState.connection then
			charmState.connection:Disconnect()
			charmState.connection = nil
		end
		charmRestoreAll()
		charmStart()
	end
end

			partOrigin.CFrame = CFrame.new(startPosition)
			partOrigin.Parent = tracerAnchor

			partTarget.Anchored = true
			partTarget.CastShadow = false
			partTarget.CFrame = CFrame.new(endPosition)
			partTarget.Parent = tracerAnchor

			local attachOrigin = Instance.new("Attachment", partOrigin)
			local attachTarget = Instance.new("Attachment", partTarget)

			local beam = Instance.new("Beam", partOrigin)
			beam.Attachment0 = attachOrigin
			beam.Attachment1 = attachTarget
			beam.Color = ColorSequence.new({
				ColorSequenceKeypoint.new(0, color1),
				ColorSequenceKeypoint.new(1, color2),
			})
			beam.Width0 = beamWidth
			beam.Width1 = beamWidth
			beam.FaceCamera = true
			beam.Transparency = NumberSequence.new(0)
			beam.LightEmission = 1

			if textureId then
				beam.Texture = "rbxassetid://" .. tostring(textureId)
				beam.TextureMode = Enum.TextureMode.Stretch
			end

			task.wait(lifetime)
			pcall(function() partOrigin:Destroy() end)
			pcall(function() partTarget:Destroy() end)
		end)
	end
end

local targetSilentAim = nil
local targetAimbot = nil
local targetRagebot = nil
local lastTargetSearchTime = 0

local function checkTargetMode(hitboxes, toggle, bestDist, currentTarget, wallCheck, teamCheck, maxFov, useFov, charFolder, isEnemy, screenCenter)
	if not toggle or not toggle.Value then
		return bestDist, currentTarget
	end
	if teamCheck and teamCheck.Value and not isEnemy then
		return bestDist, currentTarget
	end
	for _, partName in ipairs(hitboxes) do
		local part = charFolder:FindFirstChild(partName)
		if part and part:IsA("BasePart") then
			local screenPos, onScreen = currentCamera:WorldToViewportPoint(part.Position)
			if onScreen and screenPos.Z > 0 then
				if wallCheck and wallCheck.Value then
					if not isVisible(part) then
						continue
					end
				end
				local maxDistance = (useFov and useFov.Value) and maxFov or 999999
				local magnitude = (Vector2.new(screenPos.X, screenPos.Y) - screenCenter).Magnitude
				if magnitude <= maxDistance and magnitude < bestDist then
					bestDist = magnitude
					currentTarget = part
				end
			end
		end
	end
	return bestDist, currentTarget
end

FindAllTargets = function()
	local now = os.clock()
	if now - lastTargetSearchTime < 0.05 then
		return
	end
	lastTargetSearchTime = now

	if not charactersFolder then
		charactersFolder = Workspace:FindFirstChild("Characters")
		if not charactersFolder then
			return
		end
	end

	local screenCenter = currentCamera.ViewportSize / 2
	local bestAimbotDist = math.huge
	local bestAimbotTarget = nil
	local bestSilentDist = math.huge
	local bestSilentTarget = nil
	local bestRageDist = math.huge
	local bestRageTarget = nil

	for _, player in ipairs(Players:GetPlayers()) do
		if player ~= localPlayer then
			local charFolder = getCharacterFolder(player)
			if charFolder and charFolder.Parent and isAlive(charFolder) then
				local isEnemy = isEnemyPlayer(player)

				bestAimbotDist, bestAimbotTarget = checkTargetMode(
					aimbotHitboxes,
					lib.Toggles.Aimbot,
					bestAimbotDist,
					bestAimbotTarget,
					lib.Toggles.AimbotWallCheck,
					lib.Toggles.AimbotTeamCheck,
					lib.Options.AimbotFovCircleRadius and lib.Options.AimbotFovCircleRadius.Value or 150,
					lib.Toggles.AimbotUseFovCircle,
					charFolder,
					isEnemy,
					screenCenter
				)

				bestSilentDist, bestSilentTarget = checkTargetMode(
					silentAimHitboxes,
					lib.Toggles.SilentAim,
					bestSilentDist,
					bestSilentTarget,
					{
						Value = not (lib.Toggles.SilentWallbang and lib.Toggles.SilentWallbang.Value)
							or (lib.Toggles.SilentVisibleCheck and lib.Toggles.SilentVisibleCheck.Value),
					},
					lib.Toggles.SilentTeamCheck,
					lib.Options.SilentFovCircleRadius and lib.Options.SilentFovCircleRadius.Value or 150,
					lib.Toggles.SilentUseFovCircle,
					charFolder,
					isEnemy,
					screenCenter
				)

				bestRageDist, bestRageTarget = checkTargetMode(
					ragebotHitboxes,
					lib.Toggles.Ragebot,
					bestRageDist,
					bestRageTarget,
					lib.Toggles.RagebotWallCheck,
					lib.Toggles.RagebotTeamCheck,
					999999,
					{ Value = false },
					charFolder,
					isEnemy,
					screenCenter
				)
			end
		end
	end

	targetSilentAim = bestSilentTarget
	targetAimbot = bestAimbotTarget
	targetRagebot = bestRageTarget
end

	CalculateBox = function(entry)
	local charFolder = entry.CharFolder
	if not charFolder or not charFolder.Parent then
		return nil, nil, nil, nil, false
	end
	local rootPart = entry.RootPart or GetRootPart(charFolder)
	if not rootPart or not rootPart.Parent then
		return nil, nil, nil, nil, false
	end
	entry.RootPart = rootPart

	local ok, rootPos, onScreen = pcall(function()
		return currentCamera:WorldToViewportPoint(rootPart.Position)
	end)

	if not ok or not onScreen or rootPos.Z <= 0 then
		return nil, nil, nil, nil, false
	end

	if lib.Toggles.ESPDynamicBox and lib.Toggles.ESPDynamicBox.Value then
		local maxX = -math.huge
		local maxY = -math.huge
		local minX = math.huge
		local minY = math.huge
		local hasPoints = false

		for _, child in ipairs(charFolder:GetChildren()) do
			if child and child:IsA("BasePart") then
				local partPos, partOnScreen = currentCamera:WorldToViewportPoint(child.Position)
				if partOnScreen and partPos.Z > 0 then
					local cf = child.CFrame
					local sz = child.Size
					local halfX = sz.X * 0.5
					local halfY = sz.Y * 0.5
					local halfZ = sz.Z * 0.5
					local rVec = cf.RightVector
					local uVec = cf.UpVector
					local lVec = cf.LookVector
					local scale = cachedFovFactor / partPos.Z
					local extentX = (math.abs(rVec.X * halfX) + math.abs(uVec.X * halfY) + math.abs(lVec.X * halfZ)) * scale
					local extentY = (math.abs(rVec.Y * halfX) + math.abs(uVec.Y * halfY) + math.abs(lVec.Y * halfZ)) * scale

					if partPos.X - extentX < minX then minX = partPos.X - extentX end
					if partPos.X + extentX > maxX then maxX = partPos.X + extentX end
					if partPos.Y - extentY < minY then minY = partPos.Y - extentY end
					if partPos.Y + extentY > maxY then maxY = partPos.Y + extentY end
					hasPoints = true
				end
			end
		end

		if hasPoints and maxX > minX and maxY > minY then
			return maxX - minX + 4, maxY - minY + 4, minX - 2, minY - 2, true
		end
	end

	local rootHeight = rootPart.Size.Y * cachedViewportHeight / rootPos.Z * 2
	local boxWidth = 3.2 * rootHeight
	local boxHeight = 4.6 * rootHeight
	return boxWidth, boxHeight, rootPos.X - boxWidth * 0.5, rootPos.Y - boxHeight * 0.5, onScreen
end

local espTargetEntries = {}

GetEquippedWeapon = function(player)
	if not player then
		return "None"
	end

	local ok, result = pcall(function()
		return player:GetAttribute("CurrentEquipped")
	end)

	if not ok or not result or result == "" then
		return "None"
	end
	local data = nil

	pcall(function()
		data = HttpService:JSONDecode(result)
	end)

	if data and data.Name and data.Name ~= "" then
		return data.Name
	end
	return "None"
end

AddEspTarget = function(player)
	if player == localPlayer then
		return
	end

	if espTargetEntries[player] then
		RemoveEspTarget(player)
	end

	local entryData = {
		Player = player,
		CharFolder = nil,
		RootPart = nil,
		Humanoid = nil,
		Conns = {},
		Objects = {},
		LastW = nil,
		LastH = nil,
		LastX = nil,
		LastY = nil,
		LastGradTop = nil,
		LastGradBot = nil,
		LastFillT1 = nil,
		LastRatio = nil,
		LastArmorRatio = nil,
		LastHealthTop = nil,
		LastHealthMid = nil,
		LastHealthBot = nil,
		LastArmorTop = nil,
		LastArmorMid = nil,
		LastArmorBot = nil,
		LastDisplayName = nil,
		LastNameColor = nil,
		LastDist = nil,
		LastDistColor = nil,
		LastWeapon = nil,
		LastWeaponColor = nil,
		LastHealthSide = nil,
		LastArmorSide = nil,
	}

	InitEspObjects(entryData)

	local function refreshCharacterFolder()
		entryData.CharFolder = getCharacterFolder(player)
		entryData.RootPart = entryData.CharFolder and GetRootPart(entryData.CharFolder) or nil
		entryData.Humanoid = entryData.CharFolder and entryData.CharFolder:FindFirstChildOfClass("Humanoid") or nil
	end

	refreshCharacterFolder()
	entryData.Conns.CharAdded = player.CharacterAdded:Connect(refreshCharacterFolder)
	entryData.Conns.CharRemoving = player.CharacterRemoving:Connect(function()
		entryData.CharFolder = nil
		entryData.RootPart = nil
		entryData.Humanoid = nil
	end)

	task.spawn(function()
		while espTargetEntries[player] do
			task.wait(1)
			if not entryData.CharFolder or not entryData.CharFolder.Parent then
				refreshCharacterFolder()
			end
		end
	end)

	espTargetEntries[player] = entryData
end

RemoveEspTarget = function(player)
	local entryData = espTargetEntries[player]
	if not entryData then
		return
	end

	for _, conn in pairs(entryData.Conns) do
		pcall(function()
			conn:Disconnect()
		end)
	end

	table.clear(entryData.Conns)

	if entryData.Objects.TargetHolder then
		entryData.Objects.TargetHolder:Destroy()
	end

	if entryData.Objects.TracerLine then
		pcall(function()
			entryData.Objects.TracerLine:Remove()
		end)
	end

	table.clear(entryData.Objects)
	espTargetEntries[player] = nil
end

local function hideEspTarget(objects)
	if objects.TargetHolder.Visible then
		objects.TargetHolder.Visible = false
	end
	if objects.TracerLine and objects.TracerLine.Visible then
		objects.TracerLine.Visible = false
	end
end

local function getBarHolder(objects, side)
	return side == "Right" and objects.RightBarHolder or objects.LeftBarHolder
end

UpdateEspTarget = function(player, entry)
	pcall(function()
		local objects = entry.Objects
		if not objects or not objects.TargetHolder then
			return
		end

		local charFolder = entry.CharFolder
		if not charFolder or not charFolder.Parent then
			charFolder = getCharacterFolder(player)
			entry.CharFolder = charFolder
			entry.RootPart = charFolder and GetRootPart(charFolder) or nil
		end

		if not charFolder or not charFolder.Parent or not isAlive(charFolder) then
			hideEspTarget(objects)
			return
		end

		if localPlayer:GetAttribute("IsSpectating") == true then
			hideEspTarget(objects)
			return
		end

		if lib.Toggles.ESPTeamCheck and lib.Toggles.ESPTeamCheck.Value and not isEnemyPlayer(player) then
			hideEspTarget(objects)
			return
		end

		local rootPart = entry.RootPart or GetRootPart(charFolder)
		if not rootPart or not rootPart.Parent then
			hideEspTarget(objects)
			return
		end
		entry.RootPart = rootPart

		local dist = math.floor((cachedCameraPosition - rootPart.Position).Magnitude)
		local maxDist = lib.Options.ESPDistance and lib.Options.ESPDistance.Value or 500
		if dist > maxDist then
			hideEspTarget(objects)
			return
		end

		local boxW, boxH, boxX, boxY, onScreen = CalculateBox(entry)
		if not onScreen or not boxW or boxW <= 0 or boxH <= 0 then
			hideEspTarget(objects)
			return
		end

		local floorW = math.floor(boxW)
		local floorH = math.floor(boxH)
		local floorX = math.floor(boxX)
		local floorY = math.floor(boxY)

		if not objects.TargetHolder.Visible then
			objects.TargetHolder.Visible = true
		end

		if entry.LastX ~= floorX or entry.LastY ~= floorY then
			objects.TargetHolder.Position = UDim2.fromOffset(floorX, floorY)
			entry.LastX = floorX
			entry.LastY = floorY
		end

		if entry.LastW ~= floorW or entry.LastH ~= floorH then
			objects.TargetHolder.Size = UDim2.fromOffset(floorW, floorH)
			entry.LastW = floorW
			entry.LastH = floorH
		end

		local topColor = lib.Options.ESPGradientTop and lib.Options.ESPGradientTop.Value or Color3.fromRGB(19, 0, 255)
		local botColor = lib.Options.ESPGradientBot and lib.Options.ESPGradientBot.Value or Color3.fromRGB(0, 0, 0)

		if lib.Toggles.ESPBoxEnabled and lib.Toggles.ESPBoxEnabled.Value then
			if lib.Toggles.ESPBoxGlow and lib.Toggles.ESPBoxGlow.Value then
				if objects.BoxGlow.ImageTransparency ~= 0.65 then
					objects.BoxGlow.ImageTransparency = 0.65
				end
				if entry.LastGradTop ~= topColor or entry.LastGradBot ~= botColor then
					objects.BoxGlowGradient.Color = ColorSequence.new({
						ColorSequenceKeypoint.new(0, topColor),
						ColorSequenceKeypoint.new(1, botColor),
					})
				end
			elseif objects.BoxGlow.ImageTransparency ~= 1 then
				objects.BoxGlow.ImageTransparency = 1
			end

			if not objects.BoxOutlineHolder.Visible then
				objects.BoxOutlineHolder.Visible = true
			end
			if not objects.BoxInlineHolder.Visible then
				objects.BoxInlineHolder.Visible = true
			end

			if entry.LastGradTop ~= topColor or entry.LastGradBot ~= botColor then
				objects.BoxInlineGradient.Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, topColor),
					ColorSequenceKeypoint.new(1, botColor),
				})
			end

			if lib.Toggles.ESPBoxFilled and lib.Toggles.ESPBoxFilled.Value then
				if not objects.BoxFill.Visible then
					objects.BoxFill.Visible = true
				end
				local fillTransparency = lib.Options.ESPFillTransparency and lib.Options.ESPFillTransparency.Value or 0.65
				if entry.LastGradTop ~= topColor or entry.LastGradBot ~= botColor then
					objects.BoxFillGradient.Color = ColorSequence.new({
						ColorSequenceKeypoint.new(0, topColor),
						ColorSequenceKeypoint.new(1, botColor),
					})
				end
				if entry.LastFillT1 ~= fillTransparency then
					objects.BoxFillGradient.Transparency = NumberSequence.new(fillTransparency)
					entry.LastFillT1 = fillTransparency
				end
			elseif objects.BoxFill.Visible then
				objects.BoxFill.Visible = false
			end

			entry.LastGradTop = topColor
			entry.LastGradBot = botColor
		else
			if objects.BoxGlow.ImageTransparency ~= 1 then
				objects.BoxGlow.ImageTransparency = 1
			end
			if objects.BoxOutlineHolder.Visible then
				objects.BoxOutlineHolder.Visible = false
			end
			if objects.BoxInlineHolder.Visible then
				objects.BoxInlineHolder.Visible = false
			end
			if objects.BoxFill.Visible then
				objects.BoxFill.Visible = false
			end
		end

		local currentHealthSide = lib.Options.ESPHealthBarSide and lib.Options.ESPHealthBarSide.Value or "Left"
		local currentArmorSide = lib.Options.ESPArmorBarSide and lib.Options.ESPArmorBarSide.Value or "Right"

		if entry.LastHealthSide ~= currentHealthSide then
			objects.HealthBarOutline.Parent = getBarHolder(objects, currentHealthSide)
			objects.HealthBarOutline.LayoutOrder = 0
			if currentHealthSide == "Left" then
				objects.HealthBarText.AnchorPoint = Vector2.new(1, 0.5)
				objects.HealthBarText.Position = UDim2.new(0, -2, 1, 0)
			else
				objects.HealthBarText.AnchorPoint = Vector2.new(0, 0.5)
				objects.HealthBarText.Position = UDim2.new(1, 2, 1, 0)
			end
			entry.LastHealthSide = currentHealthSide
			entry.LastRatio = nil
		end

		if entry.LastArmorSide ~= currentArmorSide then
			objects.ArmorBarOutline.Parent = getBarHolder(objects, currentArmorSide)
			objects.ArmorBarOutline.LayoutOrder = 1
			if currentArmorSide == "Left" then
				objects.ArmorBarText.AnchorPoint = Vector2.new(1, 0.5)
				objects.ArmorBarText.Position = UDim2.new(0, -2, 1, 0)
			else
				objects.ArmorBarText.AnchorPoint = Vector2.new(0, 0.5)
				objects.ArmorBarText.Position = UDim2.new(1, 2, 1, 0)
			end
			entry.LastArmorSide = currentArmorSide
			entry.LastArmorRatio = nil
		end

		local humanoid = entry.Humanoid or charFolder:FindFirstChildOfClass("Humanoid")
		local currentHp = humanoid and humanoid.Health > 0 and humanoid.Health or charFolder:GetAttribute("Health") or 100
		local maxHp = humanoid and humanoid.MaxHealth > 0 and humanoid.MaxHealth or charFolder:GetAttribute("MaxHealth") or 100
		local hpRatio = math.clamp(currentHp / maxHp, 0, 1)

		local currentArmor = 0
		local armorAttr = player:GetAttribute("Armor")
		if armorAttr and armorAttr ~= "" then
			pcall(function()
				local armorData = HttpService:JSONDecode(armorAttr)
				if armorData and armorData.Health then
					currentArmor = armorData.Health
				end
			end)
		end
		local armorRatio = math.clamp(currentArmor / 100, 0, 1)

		local isHealthVisible = lib.Toggles.ESPHealth and lib.Toggles.ESPHealth.Value
		local isArmorVisible = lib.Toggles.ESPArmor and lib.Toggles.ESPArmor.Value

		if isHealthVisible then
			if not objects.HealthBarOutline.Visible then
				objects.HealthBarOutline.Visible = true
			end
			if entry.LastRatio ~= hpRatio then
				objects.HealthBar.Size = UDim2.new(1, 0, hpRatio, 0)
				objects.HealthBarText.Position = currentHealthSide == "Left" and UDim2.new(0, -2, 1 - hpRatio, 0) or UDim2.new(1, 2, 1 - hpRatio, 0)
				entry.LastRatio = hpRatio
			end

			local hpTop = lib.Options.ESPHealthTop and lib.Options.ESPHealthTop.Value or Color3.fromRGB(0, 255, 0)
			local hpMid = lib.Options.ESPHealthMid and lib.Options.ESPHealthMid.Value or Color3.fromRGB(255, 170, 0)
			local hpBot = lib.Options.ESPHealthBot and lib.Options.ESPHealthBot.Value or Color3.fromRGB(255, 0, 0)

			if entry.LastHealthTop ~= hpTop or entry.LastHealthMid ~= hpMid or entry.LastHealthBot ~= hpBot then
				objects.HealthBarGradient.Color = ColorSequence.new({
					ColorSequenceKeypoint.new(0, hpTop),
					ColorSequenceKeypoint.new(0.5, hpMid),
					ColorSequenceKeypoint.new(1, hpBot),
				})
				entry.LastHealthTop = hpTop
				entry.LastHealthMid = hpMid
				entry.LastHealthBot = hpBot
			end

			objects.HealthBarText.Text = string.format("%d", math.floor(currentHp))
			if not objects.HealthBarText.Visible then
				objects.HealthBarText.Visible = true
			end
		else
			if objects.HealthBarOutline.Visible then
				objects.HealthBarOutline.Visible = false
			end
			if objects.HealthBarText.Visible then
				objects.HealthBarText.Visible = false
			end
		end

		if isArmorVisible then
			if not objects.ArmorBarOutline.Visible then
				objects.ArmorBarOutline.Visible = true
			end
			if entry.LastArmorRatio ~= armorRatio then
				objects.ArmorBar.Size = UDim2.new(1, 0, armorRatio, 0)
				objects.ArmorBarText.Position = currentArmorSide == "Left" and UDim2.new(0, -2, 1 - armorRatio, 0) or UDim2.new(1, 2, 1 - armorRatio, 0)
				entry.LastArmorRatio = armorRatio
			end

			local arTop = lib.Options.ESPArmorTop and lib.Options.ESPArmorTop.Value or Color3.fromRGB(0, 100, 255)
			local arMid = lib.Options.ESPArmorMid and lib.Options.ESPArmorMid.Value or Color3.fromRGB(0, 50, 200)
			local arBot = lib.Options.ESPArmorBot and lib.Options.ESPArmorBot.Value or Color3.fromRGB(0, 0, 150)

					ColorSequenceKeypoint.new(0.5, arMid),
					ColorSequenceKeypoint.new(1, arBot),

				entry.LastArmorMid = arMid

			if not objects.ArmorBarText.Visible then
				objects.ArmorBarText.Visible = true
			end
		else
			if objects.ArmorBarOutline.Visible then
				objects.ArmorBarOutline.Visible = false
			end
			if objects.ArmorBarText.Visible then
				objects.ArmorBarText.Visible = false
			end
		end

		local leftHolderVisible = (isHealthVisible and currentHealthSide == "Left") or (isArmorVisible and currentArmorSide == "Left")
		local rightHolderVisible = (isHealthVisible and currentHealthSide == "Right") or (isArmorVisible and currentArmorSide == "Right")

	GetBarrelPart = function()
		local model = currentCamera:FindFirstChildOfClass("Model")
		if not model then
			cachedBarrelModel = nil
			cachedBarrelPart = nil
			return nil
		end
		if cachedBarrelModel == model and cachedBarrelPart then
			return cachedBarrelPart
		end
		cachedBarrelModel = model
		cachedBarrelPart = nil
		local weapon = model:FindFirstChild("Weapon")
		if weapon then
			local weaponModel = weapon:FindFirstChild(model.Name)
			if weaponModel then
				cachedBarrelPart = weaponModel:FindFirstChild("Barrel")
			end
		end
		if not cachedBarrelPart then
			local function searchBarrel(parent)
				for _, child in ipairs(parent:GetChildren()) do
					if child.Name == "Barrel" and child:IsA("BasePart") then
						return child
					end
					local found = searchBarrel(child)
					if found then
						return found
					end
				end
				return nil
			end
			cachedBarrelPart = searchBarrel(model)
		end
		return cachedBarrelPart
	end

do
	local crosshairDrawings = {}
	for i = 1, 8 do
		crosshairDrawings[i] = Drawing.new("Line")
		crosshairDrawings[i].Visible = false
	end
	local crosshairText = Drawing.new("Text")
	crosshairText.Visible = false
	crosshairText.Center = true
	crosshairText.Outline = true
	crosshairText.Size = 13

	local function solveCrosshairOffset(angle, distance)
		return Vector2.new(math.sin(math.rad(angle)) * distance, math.cos(math.rad(angle)) * distance)
	end

	addConnection(RunService.RenderStepped:Connect(function()
		if not crosshairConfig.enabled then
			for i = 1, 8 do
				crosshairDrawings[i].Visible = false
			end
			crosshairText.Visible = false
			return
		end

		local centerPos
		if crosshairConfig.mode == "Middle" then
			centerPos = currentCamera.ViewportSize / 2
		elseif crosshairConfig.mode == "Mouse" then
			centerPos = UserInputService:GetMouseLocation()
		elseif crosshairConfig.mode == "On Target" then
			local activeTarget
			if lib.Toggles.Ragebot and lib.Toggles.Ragebot.Value and targetRagebot then
				activeTarget = targetRagebot
			elseif lib.Toggles.SilentAim and lib.Toggles.SilentAim.Value and targetSilentAim then
				activeTarget = targetSilentAim
			elseif lib.Toggles.Aimbot and lib.Toggles.Aimbot.Value and targetAimbot then
				activeTarget = targetAimbot
			end

			if activeTarget then
				local screenPos, onScreen = currentCamera:WorldToViewportPoint(activeTarget.Position)
				if onScreen then
					centerPos = Vector2.new(screenPos.X, screenPos.Y)
				else
					centerPos = currentCamera.ViewportSize / 2
				end
			else
				centerPos = currentCamera.ViewportSize / 2
			end
		else
			centerPos = currentCamera.ViewportSize / 2
		end

		local spinAngle = 0
		if crosshairConfig.spin then
			local spinSpeed = crosshairConfig.spin_speed
			local progress = -tick() * spinSpeed % crosshairConfig.spin_max / 360
			local spinStyle = crosshairConfig.spin_style
			if spinStyle == Enum.EasingStyle.Sine then
				progress = (1 - math.cos(progress * 3.1415926535897931)) / 2
			elseif spinStyle == Enum.EasingStyle.Quad then
				progress *= progress
			elseif spinStyle == Enum.EasingStyle.Cubic then
				progress = progress * progress * progress
			elseif spinStyle == Enum.EasingStyle.Quart then
				progress ^= 4
			elseif spinStyle == Enum.EasingStyle.Quint then
				progress ^= 5
			elseif spinStyle == Enum.EasingStyle.Back then
				progress = 1 + 2.70158 * (progress - 1) ^ 3 + 1.70158 * (progress - 1) ^ 2
			elseif spinStyle == Enum.EasingStyle.Elastic then
				if progress == 0 then
					progress = 0
				elseif progress == 1 then
					progress = 1
				else
					progress = -2 ^ (10 * (progress - 1)) * math.sin((progress - 1 - 0.1) * 6.2831853071795862 / 0.3)
				end
			elseif spinStyle == Enum.EasingStyle.Bounce then
				if progress < 0.36363636363636365 then
					progress = 7.5625 * progress * progress
				elseif progress < 0.72727272727272729 then
					progress = 7.5625 * (progress - 0.54545454545454541) ^ 2 + 0.75
				elseif progress < 0.90909090909090906 then
					progress = 7.5625 * (progress - 0.81818181818181823) ^ 2 + 0.9375
				else
					progress = 7.5625 * (progress - 0.95454545454545459) ^ 2 + 0.984375
				end
			end
			spinAngle = progress * 360
		end

		local length = crosshairConfig.length
		local animLength
		if crosshairConfig.resize then
			local resizeSpeed = crosshairConfig.resize_speed
			local resizeMax = crosshairConfig.resize_max
			length = crosshairConfig.resize_min + math.sin(math.rad(tick() * resizeSpeed % 180)) * resizeMax
			animLength = crosshairConfig.resize_min + crosshairConfig.resize_max
		else
			animLength = length
		end

		local clampedLength = math.max(length, 0.5)
		local clampedAnimLength = math.max(animLength, 0.5)

		for i = 1, 4 do
			local outlineLine = crosshairDrawings[i]
			local innerLine = crosshairDrawings[i + 4]
			local angle = (i - 1) * 90 + spinAngle
			local color = crosshairConfig.color

			outlineLine.Visible = true
			outlineLine.Color = Color3.new(0, 0, 0)
			outlineLine.From = centerPos + solveCrosshairOffset(angle, crosshairConfig.radius - 1)
			outlineLine.To = centerPos + solveCrosshairOffset(angle, crosshairConfig.radius + clampedLength + 1)
			outlineLine.Thickness = crosshairConfig.width + 1.5
			outlineLine.ZIndex = 1

			innerLine.Visible = true
			innerLine.Color = color
			innerLine.From = centerPos + solveCrosshairOffset(angle, crosshairConfig.radius)
			innerLine.To = centerPos + solveCrosshairOffset(angle, crosshairConfig.radius + clampedLength)
			innerLine.Thickness = crosshairConfig.width
			innerLine.ZIndex = 2
		end

		if crosshairConfig.text and crosshairConfig.text ~= "" then
			crosshairText.Visible = true
			crosshairText.Text = crosshairConfig.text
			crosshairText.Color = crosshairConfig.color
			crosshairText.Size = crosshairConfig.textsize
			crosshairText.Position = centerPos + Vector2.new(0, crosshairConfig.radius + clampedAnimLength + 12)
			crosshairText.ZIndex = 2
			crosshairText.Outline = true
			crosshairText.OutlineColor = Color3.new(0, 0, 0)
		else
			crosshairText.Visible = false
		end
	end))
end

		esppGetBounds = function(targetModel)
			local minBound = Vector3.new(math.huge, math.huge, math.huge)
			local maxBound = Vector3.new(-math.huge, -math.huge, -math.huge)

			for _, descendant in ipairs(targetModel:GetDescendants()) do
				if descendant:IsA("BasePart") and descendant.Transparency < 0.99 then
					local halfSize = descendant.Size * 0.5
					local partPos = descendant.Position
					local minZ = partPos.Z - halfSize.Z
					minBound = Vector3.new(math.min(minBound.X, partPos.X - halfSize.X), math.min(minBound.Y, partPos.Y - halfSize.Y), math.min(minBound.Z, minZ))
					maxBound = Vector3.new(math.max(maxBound.X, partPos.X + halfSize.X), math.max(maxBound.Y, partPos.Y + halfSize.Y), math.max(maxBound.Z, partPos.Z + halfSize.Z))
				end
			end

			if minBound.X == math.huge then
				return CFrame.new(), Vector3.one
			end
			return CFrame.new((minBound + maxBound) * 0.5), maxBound - minBound
		end

		esppFrameCamera = function(centerCFrame, boundsSize)
			if not espPreviewState.Camera then
				return
			end
			local maxDimension = math.max(boundsSize.X, boundsSize.Y, boundsSize.Z) * 0.6

			if maxDimension < 0.1 then
				maxDimension = 1
			end

			local cameraDistance = maxDimension / math.tan(math.rad(espPreviewState.Camera.FieldOfView * 0.5))
			espPreviewState.Camera.CFrame = centerCFrame * CFrame.Angles(-0.17453292519943295, 0.43633231299858238, 0) * CFrame.new(0, 0, cameraDistance)
			espPreviewState.RotAngle = 0
		end

		end

		esppLoadPlayerModel = function()
			esppClearModel()
			if not espPreviewState.WorldModel then
				return
			end
			local sourceAvatar = esppFetchLocalAvatar()
			if not sourceAvatar then
				return
			end
			local archivableStates = {}

			local function setArchivable(targetInstance, archivable)
				local ok, currentArchivable = pcall(function()
					return targetInstance.Archivable
				end)

				if ok and currentArchivable ~= archivable then
					archivableStates[#archivableStates + 1] = { targetInstance, currentArchivable }

					pcall(function()
						targetInstance.Archivable = archivable
					end)
				end
			end

			setArchivable(sourceAvatar, true)

			for _, descendant in ipairs(sourceAvatar:GetDescendants()) do
				setArchivable(descendant, true)
			end

			local ok, currentModel = pcall(function()
				return sourceAvatar:Clone()
			end)

			for _, state in ipairs(archivableStates) do
				pcall(function()
					state[1].Archivable = state[2]
				end)
			end

			if not ok or not currentModel then
				warn("[Z3US ESPPreview] clone failed: " .. tostring(currentModel))
				return
			end

			for _, descendant in ipairs(currentModel:GetDescendants()) do
				if descendant:IsA("Script") or descendant:IsA("LocalScript") or descendant:IsA("ModuleScript") then

			end

			end

			end)
		end

		esppUpdateESP = function()
			local esp = espPreviewState.ESP
			if not (esp and esp.TargetHolder) then
				return
			end
			local camera = espPreviewState.Camera
			if not (camera and espPreviewState.CurrentModel and espPreviewState.CurrentModel.Parent) then
				if esp.TargetHolder.Visible then
					esp.TargetHolder.Visible = false
				end
				if esp.tracer and esp.tracer.Visible then
					esp.tracer.Visible = false
				end
				return
			end

			local function hideAll()
				if esp.TargetHolder.Visible then
					esp.TargetHolder.Visible = false
				end
				if esp.tracer and esp.tracer.Visible then
					esp.tracer.Visible = false
				end
			end

			local vpfSize = espPreviewState.VPF.AbsoluteSize
			local width = vpfSize.X
			local height = vpfSize.Y

			if width <= 0 or height <= 0 then
				width = previewWidth - 12
				height = 340
			end

			local halfWidth = width * 0.5
			local halfHeight = height * 0.5
			local focalLength = halfHeight / math.tan(math.rad(camera.FieldOfView * 0.5))
			local camCFrame = camera.CFrame

			local function project(worldPoint)
				local objPoint = camCFrame:PointToObjectSpace(worldPoint)
				local depth = -objPoint.Z
				if depth <= 0.05 then
					return nil
				end
				local screenX = halfWidth + objPoint.X / depth * focalLength
				local screenY = halfHeight - objPoint.Y / depth * focalLength
				if screenX ~= screenX or screenY ~= screenY then
					return nil
				end
				return screenX, screenY
			end

		spStripBodyParts = function(arg)
			if not arg then
				return
			end
			local weapon = arg:FindFirstChild("Weapon")

			for _, descendant in ipairs(arg:GetDescendants()) do
				if descendant:IsA("BasePart") then
					local shouldStrip

					if weapon then
						shouldStrip = not descendant:IsDescendantOf(weapon)
					else
						shouldStrip = bodyPartNames[descendant.Name] == true
					end

					if shouldStrip then
						pcall(function()
							descendant.Transparency = 1
							descendant.CanQuery = false
							descendant.CanTouch = false
							descendant.CanCollide = false
							descendant.CastShadow = false
						end)
					end
				end
			end
		end

		spCountVisibleParts = function(targetModel)
			if not targetModel then
				return 0
			end
			local visiblePartsCount = 0

			for _, descendant in ipairs(targetModel:GetDescendants()) do
				if descendant:IsA("BasePart") and descendant.Transparency < 0.99 then
					visiblePartsCount += 1
				end
			end

			return visiblePartsCount
		end

		spGetBounds = function(targetModel)
			local minBound = Vector3.new(math.huge, math.huge, math.huge)
			local maxBound = Vector3.new(-math.huge, -math.huge, -math.huge)

			for _, descendant in ipairs(targetModel:GetDescendants()) do
				if descendant:IsA("BasePart") and descendant.Transparency < 0.99 then
					local halfSize = descendant.Size * 0.5
					local partPos = descendant.Position
					local minZ = partPos.Z - halfSize.Z
					minBound = Vector3.new(math.min(minBound.X, partPos.X - halfSize.X), math.min(minBound.Y, partPos.Y - halfSize.Y), math.min(minBound.Z, minZ))
					maxBound = Vector3.new(math.max(maxBound.X, partPos.X + halfSize.X), math.max(maxBound.Y, partPos.Y + halfSize.Y), math.max(maxBound.Z, partPos.Z + halfSize.Z))
				end
			end

			if minBound.X == math.huge then
				return CFrame.new(), Vector3.one
			end
			return CFrame.new((minBound + maxBound) * 0.5), maxBound - minBound
		end

		spFrameCamera = function(centerCFrame, boundsSize)
			if not skinPreviewState.Camera then
				return
			end
			local maxDimension = math.max(boundsSize.X, boundsSize.Y, boundsSize.Z) * 0.6

			if maxDimension < 0.1 then
				maxDimension = 1
			end

			local cameraDistance = maxDimension / math.tan(math.rad(skinPreviewState.Camera.FieldOfView * 0.5))
			skinPreviewState.Camera.CFrame = centerCFrame * CFrame.Angles(-0.17453292519943295, 0.43633231299858238, 0) * CFrame.new(0, 0, cameraDistance)
			skinPreviewState.RotAngle = 0
		end

		end

		spAnchorModel = function(arg)
			for _, descendant in ipairs(arg:GetDescendants()) do
				if descendant:IsA("BasePart") then
					descendant.Anchored = true
					descendant.CanCollide = false
					descendant.CastShadow = false
				end
			end
		end

		spLoadModel = function(weaponKey, skinIdentifier)
			spClearModel()
			if not skinPreviewState.WorldModel then
				return
			end
			local skinName = "Stock"
			local weaponName = weaponKey

			if skinIdentifier and skinIdentifier ~= "Default" then
				local matchWeapon, matchSkin = skinIdentifier:match("^(.+)|(.+)$")
				weaponName = matchWeapon or weaponName
				skinName = matchSkin or skinName
			end

			local clone = nil

			if Skins then
				pcall(function()
					clone = Skins.GetCameraModel(weaponName, skinName, 0)
				end)

				if clone then
					spStripBodyParts(clone)

					if spCountVisibleParts(clone) == 0 then
						pcall(function()
							clone:Destroy()
						end)

						clone = nil
					end
				end
			end

			local needsFallback = not clone

			if needsFallback then
				pcall(function()
					local weaponSource = weaponsFolder:FindFirstChild(weaponKey) or weaponsFolder:FindFirstChild(weaponName)
					if not weaponSource then
						return
					end
					clone = (weaponSource:FindFirstChild("Camera") or weaponSource:FindFirstChild("Character") or weaponSource):Clone()

					if not clone:IsA("Model") then
						local newModel = Instance.new("Model")

						for _, child in ipairs(clone:GetChildren()) do
							child.Parent = newModel
						end

						clone:Destroy()
						clone = newModel
					end

					spStripBodyParts(clone)

					if skinIdentifier and skinIdentifier ~= "Default" and weaponName and skinName ~= "Stock" then
						local skinFolder = skinsFolder:FindFirstChild(weaponName)
						local skinVariant = skinFolder and skinFolder:FindFirstChild(skinName)
						local wearModel = skinVariant and getBestWear(skinVariant:FindFirstChild("Camera") or skinVariant)

						if wearModel then
							pcall(swapSurfaces, clone, wearModel)
						end
					end
				end)
			end

			if not clone then
				return
			end
			spAnchorModel(clone)
			clone.Parent = skinPreviewState.WorldModel
			skinPreviewState.CurrentModel = clone

			task.defer(function()
				if not skinPreviewState.CurrentModel or not skinPreviewState.CurrentModel.Parent then
					return
				end
				local boundsCenter, boundsSize = spGetBounds(skinPreviewState.CurrentModel)
				spFrameCamera(boundsCenter, boundsSize)
			end)
		end
	end

	SkinPreviewRefresh = function(text, arg)
		if not skinPreviewState.Popout then
			return
		end

		if not (lib.Toggles.SkinPreviewEnabled and lib.Toggles.SkinPreviewEnabled.Value) then
			return
		end

		if not text then
			local model = currentCamera:FindFirstChildOfClass("Model")
			text = model and model.Name or nil
			if not text or ignoredWeaponTypes[text] then
				return
			end
			arg = selectedSkins[text] or selectedSkins[getCurrentKnifeSlot()] or "Default"
		end

		local lastKey = text .. "|" .. tostring(arg)
		if lastKey == skinPreviewState.LastKey then
			return
		end
		skinPreviewState.LastKey = lastKey

		pcall(function()
			if skinPreviewState.WeaponLabel then
				skinPreviewState.WeaponLabel.Text = text
			end

			if skinPreviewState.SkinLabel then
				skinPreviewState.SkinLabel.Text = arg and arg ~= "Default" and (arg:match("^.+|(.+)$") or arg) or "Stock"
			end
		end)

		task.spawn(function()
			pcall(spLoadModel, text, arg)
		end)
	end

		local cachedPatternWeapon = nil
		local cachedPatternModel = nil

		getRCDelta = function(weaponName, shotIndex)
			if shotIndex <= 1 then
				return Vector2.zero
			end
			local sprayPatterns = customWeaponsFolder:FindFirstChild("SprayPatterns") or ReplicatedStorage.Database.Custom.Weapons:FindFirstChild("SprayPatterns")
			if not sprayPatterns then
				return Vector2.zero
			end

			if cachedPatternWeapon ~= weaponName then
				cachedPatternModel = sprayPatterns:FindFirstChild(weaponName)
				cachedPatternWeapon = weaponName
			end

			if not cachedPatternModel then
				return Vector2.zero
			end
			local currentShot = cachedPatternModel:FindFirstChild(tostring(shotIndex))
			local prevShot = cachedPatternModel:FindFirstChild(tostring(shotIndex - 1))
			if not currentShot or not prevShot then
				return Vector2.zero
			end
			local pos1 = currentShot.WorldPosition
			local pos2 = prevShot.WorldPosition
			return Vector2.new(pos1.X - pos2.X, pos1.Y - pos2.Y)
		end

		local recoilShotCount = 0
		local recoilLastShotTime = 0
		local recoilResetThreshold = 0.8

		addConnection(RunService.RenderStepped:Connect(function()
			if not (lib.Toggles.RecoilControlEnabled and lib.Toggles.RecoilControlEnabled.Value) then
				return
			end

			if recoilShotCount <= 1 then
				return
			end

			pcall(function()
				if recoilResetThreshold < os.clock() - recoilLastShotTime then
					recoilShotCount = 0
					return
				end
				local model = currentCamera:FindFirstChildOfClass("Model")
				if not model then
					return
				end
				local strength = lib.Options.RecoilControlStrength and lib.Options.RecoilControlStrength.Value or 0.5
				local rcDelta = getRCDelta(model.Name, recoilShotCount)
				if rcDelta == Vector2.zero then
					return
				end
				local screenScale = currentCamera.ViewportSize.Y / 800 * 3
				local moveX = math.clamp(-rcDelta.X * strength * screenScale, -25, 25)
				local moveY = math.clamp(rcDelta.Y * strength * screenScale, -25, 25)
				if math.abs(moveX) < 0.05 and math.abs(moveY) < 0.05 then
					return
				end

				pcall(function()
					mousemoverel(moveX, moveY)
				end)
			end)
		end))

		pcall(function()
			for _, shootObj in ipairs(getgc(true)) do
				if type(shootObj) == "table" and rawget(shootObj, "shoot") then
					if typeof(shootObj.shoot) == "function" and #debug.getupvalues(shootObj.shoot) == 26 then
						local originalShoot = nil

						local function hookShoot(self, ...)
							local shootResult = originalShoot(self, ...)

							pcall(function()
								if not (lib.Toggles.RecoilControlEnabled and lib.Toggles.RecoilControlEnabled.Value) then
									return
								end
								local now = os.clock()

								if now - recoilLastShotTime > recoilResetThreshold then
									recoilShotCount = 0
								end

								recoilShotCount += 1
								recoilLastShotTime = now
								local model = currentCamera:FindFirstChildOfClass("Model")

								if model and cachedPatternWeapon ~= model.Name then
									cachedPatternModel = nil
									cachedPatternWeapon = nil
								end
							end)

							return shootResult
						end

						originalShoot = hookfunction(shootObj.shoot, hookShoot)
						break
					end
				end
			end
		end)

		addConnection(RunService.RenderStepped:Connect(function()
			if not (lib.Toggles.Aimbot and lib.Toggles.Aimbot.Value) then
				return
			end

			if not (lib.Options.AimbotHoldkey and lib.Options.AimbotHoldkey:GetState()) then

			if not targetAimbot then

			local screenPos, onScreen = currentCamera:WorldToViewportPoint(targetAimbot.Position)
			if not onScreen or screenPos.Z <= 0 then
				return
			end
			local deltaX = screenPos.X - currentCamera.ViewportSize.X / 2
			local deltaY = screenPos.Y - currentCamera.ViewportSize.Y / 2
			local smoothness = lib.Options.AimbotSmoothness and lib.Options.AimbotSmoothness.Value or 0.5
			local factor = smoothness * smoothness * 0.8
			local moveX = deltaX * factor
			local moveY = deltaY * factor
			if math.abs(moveX) < 0.5 and math.abs(moveY) < 0.5 then
				return
			end
			mousemoverel(moveX, moveY)
		end))

			while true do
				task.wait(lib.Options.TriggerbotDelay and lib.Options.TriggerbotDelay.Value or 0)

				if lib.Toggles.Triggerbot and lib.Toggles.Triggerbot.Value then
					local mouse = localPlayer:GetMouse()

					if mouse and mouse.Target then
						local model = mouse.Target:FindFirstAncestorOfClass("Model")

						if model then
							local playerFromCharacter = nil

							if charactersFolder then
								if model.Parent == charactersFolder then
									playerFromCharacter = Players:FindFirstChild(model.Name)
								end
							end

							playerFromCharacter = playerFromCharacter or Players:GetPlayerFromCharacter(model)

							if playerFromCharacter and playerFromCharacter ~= localPlayer then
								local targetCharFolder = GetCharFolder(playerFromCharacter)

								if targetCharFolder and IsAlive(targetCharFolder) and IsEnemyPlayer(playerFromCharacter) then
									if currentEquippedWeapon and currentEquippedWeapon.IsEquipped and currentEquippedWeapon.Rounds > 0 then
										currentEquippedWeapon:shoot()

									end
									continue
								end

								continue
							end

							continue
						end
					end
				end

		end)

		pcall(function()
			local originalSend = nil

			local function hookSend(...)
				local packetArgs = { ... }
				local firstPacket = packetArgs[1]

				if firstPacket and firstPacket.Bullets and firstPacket.Bullets[1] and firstPacket.Bullets[1].Hits and firstPacket.Bullets[1].Hits[1] then
					if lib.Toggles.Ragebot and lib.Toggles.Ragebot.Value and targetRagebot then
						firstPacket.Bullets[1].Hits[1].Instance = targetRagebot
						firstPacket.Bullets[1].Hits[1].Position = targetRagebot.Position
					end

					if lib.Toggles.SilentAim and lib.Toggles.SilentAim.Value and targetSilentAim then
						local hitchance = lib.Options.SilentHitchance and lib.Options.SilentHitchance.Value or 100

						if math.random(1, 100) <= hitchance then
							firstPacket.Bullets[1].Hits[1].Instance = targetSilentAim
							firstPacket.Bullets[1].Hits[1].Position = targetSilentAim.Position
						end
					end
				end

				if firstPacket and firstPacket.Bullets then
					local camPos = currentCamera.CFrame.Position

					for _, bullet in ipairs(firstPacket.Bullets) do
						local hitPos = nil
						local hitInstance = nil

						if bullet.Hits and #bullet.Hits > 0 then
							local firstHit = bullet.Hits[1]
							hitPos = firstHit.Position
							hitInstance = firstHit.Instance
						end

						if not hitPos then
							local lookVector = currentCamera.CFrame.LookVector
							local raycastParams = RaycastParams.new()
							raycastParams.FilterType = Enum.RaycastFilterType.Exclude
							local localCharFolder = GetCharFolder(localPlayer)
							raycastParams.FilterDescendantsInstances = localCharFolder and { localCharFolder } or {}
							local rayResult = Workspace:Raycast(camPos, lookVector * 500, raycastParams)
							hitPos = rayResult and rayResult.Position or camPos + lookVector * 500
							hitInstance = rayResult and rayResult.Instance or nil
						end

						if lib.Toggles.HitsoundsEnabled and lib.Toggles.HitsoundsEnabled.Value then
							local soundName = lib.Options.HitsoundsSound and lib.Options.HitsoundsSound.Value or "None"

							if soundName ~= "None" then
								local isBasePart = hitInstance and hitInstance:IsA("BasePart")
								local isEnemyHit = false

								if isBasePart then
									local hitModel = hitInstance:FindFirstAncestorOfClass("Model")

									if hitModel then
										local hitPlayer = Players:GetPlayerFromCharacter(hitModel)

										if hitPlayer and hitPlayer ~= localPlayer and IsEnemyPlayer(hitPlayer) then
											isEnemyHit = true
										end
									end
								end

								if isEnemyHit then
									local volume = lib.Options.HitsoundsVolume and lib.Options.HitsoundsVolume.Value or 50
									local soundId = hitSoundsConfig.sounds[soundName]

									if soundId then
										task.spawn(function()
											local hitSound = Instance.new("Sound")
											hitSound.SoundId = soundId
											hitSound.Volume = math.clamp(volume / 100, 0, 1)
											hitSound.Parent = workspace
											hitSound:Play()

											hitSound.Ended:Connect(function()
												hitSound:Destroy()
											end)
										end)
									end
								end
							end
						end

						if lib.Toggles.BulletTracerEnabled and lib.Toggles.BulletTracerEnabled.Value then
							SpawnTracerBeam(camPos, hitPos)

							if lib.Toggles.BulletTracerImpacts and lib.Toggles.BulletTracerImpacts.Value then
								SpawnTracerImpact(hitPos)
							end
						end
					end
				end

				return originalSend(unpack(packetArgs))
			end

			originalSend = hookfunction(send, hookSend)
		end)

do
		local bhopState = {
			Armed = false,
			WasSpace = false,
			AirTicks = 0,
			FlipTimer = 0,
			StrafeSide = 1,
			MouseBuffer = {},
		}

		local RuntimeKinematics = nil
		local LocalCharacter = nil

		pcall(function()
			RuntimeKinematics = require(ReplicatedStorage.MovementV2.RuntimeKinematics)
			LocalCharacter = require(ReplicatedStorage.Controllers.CharacterController.LocalCharacter)
		end)

		GetBhopCharacter = function()
			if not LocalCharacter then
				return nil
			end

			local ok, result = pcall(function()
				return LocalCharacter:getCurrentCharacter()
			end)

			return ok and result or nil
		end

		ApplyBhopShadow = function(targetCharacter)
			if not targetCharacter or targetCharacter.__Bhop then
				return
			end

			local ok, originalSampleInput = pcall(function()
				return targetCharacter.SampleInput
			end)

			if not ok or type(originalSampleInput) ~= "function" then
				return
			end
			targetCharacter.__Bhop = true

			targetCharacter.SampleInput = function(self, inputState)
				local ok2, sampleResult = pcall(originalSampleInput, self, inputState)
				if not ok2 or type(sampleResult) ~= "table" then
					return nil
				end

				pcall(function()
					if not UserInputService:IsKeyDown(Enum.KeyCode.Space) then
						return
					end
					local character = localPlayer.Character
					if not character or not RuntimeKinematics then
						return
					end
					local kinematics = RuntimeKinematics.read(character)
					if not kinematics or kinematics.OnGround then
						return
					end
					sampleResult.Move = Vector2.new(bhopState.StrafeSide, 0)
				end)

				return sampleResult
			end
		end

		BhopUpdateStrafe = function()
			bhopState.FlipTimer = bhopState.FlipTimer + 1
			local mouseDelta = UserInputService:GetMouseDelta()
			local strafeThreshold = lib.Options.BhopStrafeThreshold and lib.Options.BhopStrafeThreshold.Value or 0.3
			local autoFlipTicks = lib.Options.BhopAutoFlipTicks and lib.Options.BhopAutoFlipTicks.Value or 6
			local strafeChanged = false

			if mouseDelta then
				table.insert(bhopState.MouseBuffer, mouseDelta.X)

				if #bhopState.MouseBuffer > 2 then
					table.remove(bhopState.MouseBuffer, 1)
				end

				local mouseSum = 0

				for _, deltaX in ipairs(bhopState.MouseBuffer) do
					mouseSum += deltaX
				end

				local avgMouseDelta = mouseSum / #bhopState.MouseBuffer

				if math.abs(avgMouseDelta) >= strafeThreshold then
					bhopState.StrafeSide = avgMouseDelta > 0 and 1 or -1
					bhopState.FlipTimer = 0
					strafeChanged = true
				end
			end

			for _, keyMap in ipairs({ { Side = -1, Code = Enum.KeyCode.A }, { Side = 1, Code = Enum.KeyCode.D } }) do
				if UserInputService:IsKeyDown(keyMap.Code) then
					bhopState.StrafeSide = keyMap.Side
					bhopState.FlipTimer = 0
					strafeChanged = true
				end
			end

			if not strafeChanged and bhopState.FlipTimer >= autoFlipTicks then
				bhopState.StrafeSide = bhopState.StrafeSide * -1
				bhopState.FlipTimer = 0
			end
		end

		addConnection(RunService.PreRender:Connect(function()
			if not (lib.Toggles.BhopEnabled and lib.Toggles.BhopEnabled.Value) then
				return
			end

			if not RuntimeKinematics or not LocalCharacter then
				return
			end
			local character = localPlayer.Character
			if not character then
				return
			end
			local bhopChar = GetBhopCharacter()

			if bhopChar then
				ApplyBhopShadow(bhopChar)
			end

			local kinematics = RuntimeKinematics.read(character)
			if not kinematics then
				return
			end

			if UserInputService:IsKeyDown(Enum.KeyCode.Space) and bhopChar and bhopChar.JumpInputDown then
				pcall(function()
					bhopChar:SetJumpInput(false)
				end)
			end

			local isSpaceDown = UserInputService:IsKeyDown(Enum.KeyCode.Space)

			if kinematics.OnGround then
				bhopState.AirTicks = 0

				if isSpaceDown and bhopState.Armed then
					pcall(function()
						bhopChar:SetJumpInput(nil)
					end)

					bhopState.Armed = false
				end

				if isSpaceDown and not bhopState.WasSpace then
					pcall(function()
						bhopChar:SetJumpInput(nil)
					end)
				end
			else
				bhopState.Armed = true
				bhopState.AirTicks = bhopState.AirTicks + 1
			end

			bhopState.WasSpace = isSpaceDown
			BhopUpdateStrafe()
		end))
	end
